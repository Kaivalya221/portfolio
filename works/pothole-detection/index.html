import cv2
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# 1. Load raw road-surface image frame
img = cv2.imread("road_test_frame.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 2. Filter noise and run bilateral image thresholding
blur = cv2.bilateralFilter(gray, 9, 75, 75)
thresh = cv2.adaptiveThreshold(blur, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                               cv2.THRESH_BINARY_INV, 11, 2)

# 3. Detect candidate pothole contours and calculate areas
contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
potholes = []

for cnt in contours:
    area = cv2.contourArea(cnt)
    if 500 < area < 50000: # Filter noise from actual potholes
        potholes.append(cnt)
        x, y, w, h = cv2.boundingRect(cnt)
        cv2.rectangle(img, (x, y), (x + w, y + h), (0, 0, 255), 2)
        print(f"[DETECT] Pothole found at x={x}, y={y} with area={area}")

# 4. Generate 3D depth-map mesh from single image shading (Shape from Shading)
def reconstruct_3d_depth(roi_gray):
    """
    Maps pixel intensity gradient offsets to simulate depth variations
    """
    rows, cols = roi_gray.shape
    X = np.arange(0, cols, 1)
    Y = np.arange(0, rows, 1)
    X, Y = np.meshgrid(X, Y)
    
    # Standard shape estimation approximation: darken pixels represent deeper slots
    Z = (255.0 - roi_gray.astype(float)) / 255.0 * -15.0 # Scale depth to max -15cm
    return X, Y, Z

if len(potholes) > 0:
    x, y, w, h = cv2.boundingRect(potholes[0])
    roi = gray[y:y+h, x:x+w]
    
    X_mesh, Y_mesh, Z_depth = reconstruct_3d_depth(roi)
    
    # 5. Output metrics and plot wireframe height layout
    avg_depth = np.min(Z_depth)
    est_volume = np.sum(np.abs(Z_depth)) * 0.0001 # volume scale helper
    print(f"[METRIC] Estimated Max Depth: {avg_depth:.2f} cm")
    print(f"[METRIC] Estimated Volume: {est_volume:.4f} m3")

    fig = plt.figure()
    ax = fig.add_subplot(111, projection='3d')
    ax.plot_wireframe(X_mesh, Y_mesh, Z_depth, color='red')
    plt.savefig("pothole_depth_mesh.png")
